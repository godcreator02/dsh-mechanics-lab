一个实例最少需要什么

两个文件、一个空 bundle 列表，能撑起多少东西——以及哪根线剪了实例就死。

档次 ① 🔬 发现型 ✅ 5 条用例 不需要 web

一个实例是怎么拼出来的

一个 dsh 实例能跑起来，靠的是几处文件被按固定顺序拼成一棵插件树。最外层是 profile——一个目录，同时是一个 Node 包，也是一套组合配方；配方由几层 patch（补丁：对条目列表的一次编辑指令）按顺序叠加，叠出最终的 条目（entry：配方里的一行，写着挂哪个插件、给它什么配置）列表——这份 列表就是最终要跑起来的插件树。每条条目在运行时对应一个 fiber，也就是 这个插件这一次生命的载体。这一项要讲清楚的，是这整套拼装链条里最少需要哪些 零件：从几个文件，到一个真的在跑的进程。

/**
 * 树的普查员：从进程内部给 loader 上的整棵条目树拍快照。
 *
 * 存在的理由：`--dump-config` 只能告诉你「配方算出来是什么」，看不到框架在
 * boot 结束之后才补上的条目（那些条目不在任何 patch 文件里）。要看见它们，
 * 只能从进程内部往外看。
 *
 * 拍两张快照，这是本插件全部的设计：
 *
 *   boot   —— apply 被调用的当下。这时 boot() 还没返回，框架的补丁还没打。
 *   settle —— 延后 delayMs 之后。这时 boot() 早已返回，补的条目都到位了。
 *
 * 延后用 Node 原生 setTimeout，不用 `ctx.setTimeout` —— 后者是 timer 服务提供的，
 * 而 timer 在不在正是这一组要测的东西之一，拿被测对象当测量工具就是循环论证。
 */

import { writeFileSync } from "node:fs";

/** 代码版本指纹。 */
const MARKER = "base-census-v1";

/** 关心的服务名。DSH 的服务表很长，只探跟「最小环境」有关的这几个。 */
const WATCHED_SERVICES = ["loader", "timer", "hmr", "logger", "webServer"];

/**
 * 找一个条目的**父条目**。
 *
 * 条目树是分层的，但 `loader.entries()` 是扁平遍历（自己 + 所有嵌套子树），
 * 光看那个列表分不出谁在谁下面。取法照抄 loader 源码里 `getOuterStack` 的走法：
 * `entry.parent` 是它所属的 EntryGroup，那个组的 `ctx.fiber.entry` 就是拥有
 * 这棵子树的条目；根组没有拥有者，返回 null。
 */
function parentIdOf(entry) {
  try {
    return entry.parent?.ctx?.fiber?.entry?.options?.id ?? null;
  } catch {
    return null; // 观测点绝不抛错
  }
}

function censusEntry(entry) {
  const fiber = entry.fiber;
  return {
    id: entry.options?.id ?? null,
    name: entry.options?.name ?? null,
    parent: parentIdOf(entry),
    disabled: entry.options?.disabled === true,
    hasFiber: fiber !== undefined,
    fiberState: fiber === undefined ? null : fiber.state,
    // inject 是「这个条目在等哪些服务」。PENDING 卡住时这里就是原因。
    inject: fiber === undefined ? null : Object.keys(fiber.inject ?? {}),
  };
}

function takeSnapshot(ctx, phase) {
  const loader = ctx.get("loader");
  const services = {};
  for (const name of WATCHED_SERVICES) {
    services[name] = ctx.get(name) !== undefined;
  }
  return {
    phase,
    at: new Date().toISOString(),
    services,
    // loader 拿不到就如实记 null，绝不 throw —— 观测点抛错会把
    // 「根本没跑到这」和「跑到了但没拿到」混成同一个结果。
    entries: loader === undefined ? null : [...loader.entries()].map(censusEntry),
  };
}

/**
 * 历次 apply 的记录，**模块级**。
 *
 * 为什么必须是模块级 —— 普查员会被重挂。挂在 apply 局部变量里，重挂会从零
 * 开始并把文件整个覆盖掉，settle 快照就可能凭空消失（定时器被 dispose 清掉，
 * 新的一轮又只写了 boot 快照）——工具会把自己被重挂这件事藏住。
 *
 * 挪到模块级之后，重挂会往数组里多加一条，文件里因此看得见「这个条目被挂了
 * 几次」，从缺陷变成了一个有用的观测量。模块级变量只在**模块被重新 import**
 * 时才清空（代码热重载），条目重挂不清。
 */
const records = [];

export function apply(ctx, config) {
  const out = config?.out;
  if (!out) throw new Error("base-census: config.out 是必需的");
  const delayMs = config?.delayMs ?? 1500;

  const record = {
    marker: MARKER,
    applyIndex: records.length,
    delayMs,
    snapshots: [takeSnapshot(ctx, "boot")],
  };
  records.push(record);

  // 写整个数组：读的一方永远看得到全部历史，不用担心被后一次覆盖
  const flush = () => writeFileSync(out, JSON.stringify(records, null, 2), "utf8");
  flush();

  const handle = setTimeout(() => {
    record.snapshots.push(takeSnapshot(ctx, "settle"));
    flush();
  }, delayMs);
  // 不 unref：这个定时器要能撑住事件循环——本组有一问就是「谁保持进程活着」，
  // 而普查员自己必须**不**成为那个答案，所以它在 flush 完就把定时器清掉，
  // 之后进程还活不活着，就完全是别人的事了。
  ctx.on("dispose", () => clearTimeout(handle));
}

{
  "name": "lab-recorder",
  "version": "0.1.0",
  "private": true,
  "description": "观测台的采集器：订阅 Cordis 生命周期事件，追加写 jsonl。无 UI、零 inject。",
  "type": "module",
  "exports": { ".": "./index.js" }
}

/**
 * lab-recorder — 观测台的采集器。
 *
 * 唯一必须住在被观测进程里的东西：订阅 Cordis 与 loader 的生命周期事件，
 * 追加写 jsonl。看板是独立的 Python 服务，读这些文件。
 *
 * ── 事件分三类处理，依据是**是不是 waterfall**，不是"量大不大" ──────────
 *
 *   流水型  低频、有时序意义 → 全量记进 events.jsonl，时间线展示
 *   关系型  价值在去重后的集合 → 聚合进 relations.json，不进时间线
 *   链路型  waterfall：监听器**必须调 next()**，否则阻断整条执行链。
 *           订阅它等于插进执行路径，不再是只读观测 → **默认关**
 *
 * 为什么关系型要去重：`internal/get` 每次服务访问都触发，热路径上每秒几千次。
 * 记流水的话一次运行几万条、几十 MB，而"webServer 被读了 8000 次"毫无价值；
 * 去重成「谁读过什么服务」之后只剩几十条，还直接画得出服务依赖图。
 *
 * 为什么 waterfall 默认关：L10 / L13 测的就是时序。订阅 waterfall 会让每次
 * config 更新、每次服务读写都多走一层回调，next() 返回 Promise 时还会改变异步
 * 调度。仪器插在执行链里会让那两课的结论可疑。平时想看依赖细节再打开。
 *
 * ── 三条设计约束 ────────────────────────────────────────────────────────
 *
 * 1. **零 inject**。订阅只要 apply 的 ctx，写文件只要 node:fs。没有依赖就没有
 *    等待，它会在很早的批次挂上，漏掉的早期事件最少。
 *    （反面教材：lab-inspector 因为 inject 了 loader，被 loader 的 await 语义
 *    锁死在 PENDING —— 见 demo/README.md。）
 * 2. **内存缓冲 + 定时 flush**。逐个同步写盘会明显改变时序，而我们测的是
 *    "什么变了什么没变"这类定性结论，宁可丢掉最后 250ms 也不污染时序。
 * 3. **只记录，不判断**。事件原样落盘，过滤/聚合/解释全在看板做。
 *    采集器越笨越好 —— 它是唯一会影响被观测系统的部分。
 */

import { appendFileSync, mkdirSync, writeFileSync } from "node:fs";
import { dirname } from "node:path";

/**
 * FiberState 的数字 → 名字。
 *
 * 必须自己写：`FiberState` 在 cordis 里是 TypeScript 的 `const enum`，
 * 编译后被内联成数字字面量，**运行时不存在这个对象**，import 不到。
 */
const FIBER_STATE = ["PENDING", "LOADING", "ACTIVE", "FAILED", "DISPOSED", "UNLOADING"];

const MARKER = "lab-recorder-v2";

/** config 之类的任意值记进日志前先压扁，防止一条事件撑爆整个文件。 */
function brief(value, max = 400) {
  if (value === undefined) return undefined;
  try {
    const text = JSON.stringify(value);
    if (text === undefined) return String(value);
    return text.length > max ? text.slice(0, max) + `…(共${text.length}字符)` : JSON.parse(text);
  } catch {
    return String(value);
  }
}

export function apply(ctx, config) {
  const out = config?.out;
  if (!out) throw new Error("lab-recorder: config.out（事件日志路径）是必需的");

  const relOut = config?.relationsOut ?? out.replace(/\.jsonl$/, "") + ".relations.json";
  const flushMs = config?.flushMs ?? 250;
  /** 是否订阅 waterfall 事件。**默认关** —— 理由见文件头。 */
  const waterfall = config?.waterfall === true;

  mkdirSync(dirname(out), { recursive: true });
  // 每次挂载重开：一次运行一份日志，避免跨运行串味
  writeFileSync(out, "", "utf8");

  const t0 = process.hrtime.bigint();
  const buffer = [];
  let seq = 0;

  const push = (kind, payload) => {
    buffer.push({
      seq: seq++,
      // 相对采集器挂载时刻的毫秒数，三位小数 —— 看得出几毫秒的窄窗口
      ms: Number(process.hrtime.bigint() - t0) / 1e6,
      at: new Date().toISOString(),
      kind,
      ...payload,
    });
  };

  /** 记录时绝不能抛：采集器把被观测系统搞崩是最坏的结果。 */
  const safely = (fn) => {
    try {
      fn();
    } catch {
      /* 观测失败就观测失败，绝不影响被观测者 */
    }
  };

  // ── 关系表（去重聚合，不进时间线）────────────────────────────────────
  /** key = "谁\0什么"，value = {who, what, count} */
  const serviceReads = new Map();
  const listeners = new Map();
  /** 当前存在的服务：name → 最近一次绑定是否有值 */
  const services = new Map();

  const bump = (map, who, what) => {
    const key = who + "" + what;
    const hit = map.get(key);
    if (hit) hit.count += 1;
    else map.set(key, { who, what, count: 1 });
  };

  /** 从一个 Context 反查它属于哪个条目 —— 关系表的「谁」。 */
  const whoOf = (someCtx) => {
    const entry = someCtx?.fiber?.entry;
    return entry?.options?.id ?? entry?.options?.name ?? "(无条目)";
  };

  /** 从 fiber 反查条目。不是所有 fiber 都对应条目（插件内部 ctx.plugin() 起的没有）。 */
  const describeFiber = (fiber) => {
    const entry = fiber?.entry;
    if (entry?.options) {
      return { scope: "entry", id: entry.options.id ?? null, name: entry.options.name ?? null };
    }
    return { scope: "inner", id: null, name: fiber?.runtime?.name ?? null };
  };

  const describeEntry = (entry) => ({
    scope: "entry",
    id: entry?.options?.id ?? null,
    name: entry?.options?.name ?? null,
  });

  // ══ 对外服务：让被观测的插件自己说话 ══════════════════════════════════
  //
  // 采集器订阅的那些事件回答的是「框架对这个条目做了什么」，回答不了
  // 「这个插件自己拿到了什么、干了什么」—— 那些只有插件自己知道。
  //
  // 所以再开一个口子：插件 `inject: ["labObserver"]`，把自己的话说进同一条
  // 事件流。好处是上报与框架事件**时序天然对齐**，看板不用去拼两份数据；
  // 教学插件也不必各自 writeFileSync 一份格式各异的见证文件。
  //
  // 为什么是 `for(ctx)` 而不是直接 `say(...)`：服务方法拿不到调用方的
  // context，而「谁说的」不能靠插件自报（它写死的名字未必等于条目 id）。
  // 传一次自己的 ctx，采集器用现成的 whoOf 反查真实身份，比自报可靠。
  const observer = {
    /** 绑定调用者身份，返回一个上报函数。 */
    for(callerCtx) {
      const who = whoOf(callerCtx);
      return (note, data) =>
        safely(() => push("report", { who, note: String(note), data: brief(data) }));
    },
  };

  // provide 本身就是个 effect：fiber 卸载时服务自动撤销，不用自己包 ctx.effect
  ctx.provide("labObserver", observer);

  // ══ 流水型：全量记录 ══════════════════════════════════════════════════

  // fiber 生命周期状态每变一次触发一次，带旧状态。
  // 这是能抓住 LOADING 这种几毫秒窄窗口的唯一办法 —— 轮询必然错过。
  ctx.on("internal/status", (fiber, oldValue) =>
    safely(() => push("status", {
      ...describeFiber(fiber),
      from: FIBER_STATE[oldValue] ?? String(oldValue),
      to: FIBER_STATE[fiber?.state] ?? String(fiber?.state),
    })));

  // fiber 被创建，或销毁时 uid 被清空
  ctx.on("internal/plugin", (fiber) =>
    safely(() => push("plugin", { ...describeFiber(fiber), uid: fiber?.uid ?? null })));

  // 条目被重配置或部分销毁。**legacy 是改动前的 options** —— 能直接看出
  // 「从什么改成了什么」，是 L10 最有价值的一条。
  ctx.on("loader/partial-dispose", (entry, legacy, active) =>
    safely(() => push("entry-dispose", {
      ...describeEntry(entry),
      active,
      legacy: brief(legacy),
    })));

  ctx.on("loader/entry-init", (entry) =>
    safely(() => push("entry-init", describeEntry(entry))));

  ctx.on("loader/config-update", () =>
    safely(() => push("config-update", { note: "loader 把当前树写回了磁盘" })));

  // 服务绑定变化：value 非空＝出现，空＝撤销。L3 的核心观测。
  ctx.on("internal/service", function (name, value) {
    safely(() => {
      const present = value !== undefined && value !== null;
      services.set(name, present);
      push("service", { serviceName: name, present, who: whoOf(this) });
    });
  });

  // hmr 三件套
  ctx.on("hmr/change", (url) =>
    safely(() => push("hmr-change", { url: String(url) })));

  ctx.on("hmr/reload", (reloads) =>
    safely(() => {
      // reloads 是 Map<Plugin, Reload>，key 是插件函数，只能取 Reload.filename
      const files = [];
      try {
        for (const [, reload] of reloads) files.push(reload?.filename ?? null);
      } catch { /* 不是 Map 就算了 */ }
      push("hmr-reload", { count: files.length, files: brief(files, 800) });
    }));

  ctx.on("hmr/config-update-failed", (filename, error) =>
    safely(() => push("hmr-failed", {
      filename: String(filename),
      error: String(error?.message ?? error),
    })));

  // ══ 关系型：去重聚合 ══════════════════════════════════════════════════
  // internal/listener 不是 waterfall（签名里没有 next），可以安全订阅。
  ctx.on("internal/listener", function (name, listener, prepend) {
    safely(() => bump(listeners, whoOf(this), String(name)));
  });

  // ══ 链路型：waterfall，默认关 ═════════════════════════════════════════
  // ⚠️ 这些监听器**必须调用 next() 并传回它的返回值**，否则阻断整条执行链。
  // 订阅它们不再是只读观测，而是插进了执行路径 —— 所以默认不开。
  if (waterfall) {
    push("recorder", { marker: MARKER, note: "⚠️ waterfall 观测已开启：仪器插在执行链里，测时序的实验请关掉" });

    // 每次服务访问都触发，热路径。只做去重聚合，绝不记流水。
    ctx.on("internal/get", function (someCtx, name, error, next) {
      safely(() => bump(serviceReads, whoOf(someCtx), String(name)));
      return next();
    });

    ctx.on("internal/set", function (someCtx, name, value, error, next) {
      safely(() => push("service-set", { serviceName: String(name), who: whoOf(someCtx) }));
      return next();
    });

    ctx.on("internal/config", function (cfg, next) {
      safely(() => push("config-resolve", { ...describeFiber(this), config: brief(cfg) }));
      return next();
    });

    ctx.on("internal/update", function (cfg, noSave, next) {
      safely(() => push("config-update-fiber", { ...describeFiber(this), noSave, config: brief(cfg) }));
      return next();
    });

    ctx.on("loader/patch-context", function (entry, next) {
      safely(() => push("patch-context", describeEntry(entry)));
      return next();
    });
  }

  // ── flush ───────────────────────────────────────────────────────────────
  const flushRelations = () => {
    const payload = {
      at: new Date().toISOString(),
      marker: MARKER,
      waterfall,
      services: [...services.entries()].map(([name, present]) => ({ name, present })),
      listeners: [...listeners.values()],
      serviceReads: [...serviceReads.values()],
    };
    try {
      writeFileSync(relOut, JSON.stringify(payload, null, 2), "utf8");
    } catch { /* 同上，观测失败不影响被观测者 */ }
  };

  const flush = () => {
    if (buffer.length > 0) {
      const lines = buffer.map((e) => JSON.stringify(e)).join("\n") + "\n";
      buffer.length = 0;
      try {
        appendFileSync(out, lines, "utf8");
      } catch { /* 同上 */ }
    }
    flushRelations();
  };

  const timer = setInterval(flush, flushMs);
  // unref：别让这个定时器把进程钉住、导致实例该退出时退不出去
  timer.unref?.();

  push("recorder", {
    marker: MARKER,
    waterfall,
    note: "采集开始（本行之前的事件未被记录）",
  });

  // ── 起点快照 ────────────────────────────────────────────────────────────
  // 采集器自己也是树上的一个条目，在它 apply 之前挂载的条目，其
  // PENDING→LOADING 已经发生完了（实测：lab-minimal 只赶上最后一跳）。
  // 而加载顺序由**服务可用性**驱动、不由书写顺序决定，所以没法把自己排到第一个。
  //
  // 补救：订阅之后立刻拍一张全树快照，记下每个条目此刻的状态。
  // 丢掉的仍然是**过程**，但起点是确切的。
  //
  // 用 ctx.get 运行时取 loader，**不写进 inject** —— loader 的 intercept 有
  // await 语义，声明依赖它会把本插件锁死在 PENDING（见 demo/README.md）。
  // 顺序上先订阅后快照：宁可重复也不漏。
  safely(() => {
    const loader = ctx.get("loader");
    if (!loader?.entries) {
      push("recorder", { marker: MARKER, note: "拿不到 loader，跳过起点快照" });
      return;
    }
    let count = 0;
    // 统计「睁眼时已经在半路上的」——**不是**「已经 ACTIVE 的」。
    // 采集器零 inject、在第一批就挂上，实测 81 个条目里已 ACTIVE 的是 0 个，
    // 按那个口径自述会读成「什么都没丢」，与事实相反：真正丢了头一跳的是
    // 快照时已经建起来（state 非 null）的那几条 —— 其中就包括采集器自己。
    let midway = 0;
    for (const entry of loader.entries()) {
      const options = entry?.options ?? {};
      const fiber = entry?.fiber;
      const state = fiber ? FIBER_STATE[fiber.state] ?? String(fiber.state) : null;
      if (state !== null) midway += 1;
      push("snapshot", {
        scope: "entry",
        id: options.id ?? null,
        name: options.name ?? null,
        to: state,
        hasFiber: fiber !== undefined,
        disabled: Boolean(entry?.disabled),
        config: brief(options.config, 200),
      });
      count += 1;
    }
    // 量化「漏了多少」：快照时已经建起来的条目，它们的头一跳我们没看到
    push("recorder", {
      marker: MARKER,
      note: `起点快照完成：${count} 个条目，其中 ${midway} 个睁眼时已在半路上（它们的出生过程没记全）`,
    });
  });

  // 卸载时停表并把尾巴刷干净
  ctx.effect(() => () => {
    clearInterval(timer);
    push("recorder", { marker: MARKER, note: "采集结束" });
    flush();
  }, "lab-recorder: flush timer");
}
